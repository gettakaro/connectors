# Takaro Terraria bridge settings. The release zip ships this file as
# TakaroTerrariaBridge/TakaroConfig.txt; edit it there.
#
# Paste the registration token Takaro shows when you add a Generic game server into
# registrationToken= and save. This works before or after the bridge starts: the bridge
# reads this file every few seconds and connects on its own, no restart needed.
#
# Leave identityToken empty; the bridge fills it in. It identifies this server in Takaro,
# so keep it if you move or reinstall the server. Leave configFormat as it is.

configFormat=2

# Takaro connection
registrationToken=
identityToken=
serverName=Terraria Server
serverChatName=Takaro
takaroWsUrl=wss://connect.takaro.io/

# TShock REST API. Enable RestApiEnabled in tshock/config.json first.
tshockBaseUrl=http://127.0.0.1:7878
tshockToken=
# Alternatively, create a token from credentials:
# tshockUsername=admin
# tshockPassword=

# Local bridge HTTP server (GET /health, GET /coverage)
httpPort=3020
pollIntervalMs=10000

# TShock log files to tail for chat, join/leave, and TAKARO_EVENT markers.
# Point this at the log DIRECTORY, not a single file: TShock opens a new log on
# every restart, and a directory lets the bridge follow that rotation. A single
# file path still works but stops receiving events once the server restarts.
logFiles=tshock/logs

# Substring matches (case-insensitive, comma-separated) never shipped as `log` events.
# Defaults to `takaro-rest executed:,RestManager:` — TShock logs every REST call the bridge
# itself makes, and tailing those feeds the bridge's own traffic back to Takaro until the
# rate limiter drops real gameplay events. Set to an empty value to disable filtering.
logExcludePatterns=takaro-rest executed:,RestManager:

# Raw console commands are allowlisted by exact match and by prefix.
commandAllowlistExact=help,/help
commandAllowlistPrefixes=say,time

# Shutdown is separately gated and off by default.
enableShutdown=false
